Minimal Python Stratum client for testing and development:
import socket, json, hashlib

HOST    = 'YOUR_NODE_IP'
PORT    = 3333
ADDRESS = 'SPN1YourAddress'
WORKER  = 'python01'

def sha256d(data):
    b = bytes.fromhex(data) if isinstance(data, str) else data
    return hashlib.sha256(hashlib.sha256(b).digest()).hexdigest()

sock = socket.socket()
sock.connect((HOST, PORT))

def send(msg):
    sock.sendall((json.dumps(msg) + '\n').encode())

def recv():
    buf = b''
    while b'\n' not in buf:
        buf += sock.recv(4096)
    return json.loads(buf.split(b'\n')[0])

# Step 1: Subscribe
send({"id": 1, "method": "mining.subscribe", "params": ["python-miner/1.0"]})
sub    = recv()
extra1 = sub['result'][1]
extra2_sz = sub['result'][2]
print(f"Subscribed — ExtraNonce1: {extra1}, ExtraNonce2 size: {extra2_sz}")

# Step 2: Authorize (format: ADDRESS.WORKERNAME)
send({"id": 2, "method": "mining.authorize", "params": [f"{ADDRESS}.{WORKER}", "x"]})
auth = recv()
print(f"Authorization: {'OK' if auth['result'] else 'FAILED — check address format'}")

# Step 3: Receive jobs and submit shares
while True:
    msg = recv()
    if msg.get('method') == 'mining.set_difficulty':
        print(f"Difficulty set to: {msg['params'][0]}")
    elif msg.get('method') == 'mining.notify':
        job_id, prev, cb1, cb2, branches, ver, bits, ntime, clean = msg['params']
        extra2 = '00' * extra2_sz
        cb_hash = sha256d(cb1 + extra1 + extra2 + cb2)
        print(f"New job: {job_id} | prev: {prev[:16]}...")
        # Submit share (replace nonce with actual PoW result)
        send({"id": 3, "method": "mining.submit",
              "params": [f"{ADDRESS}.{WORKER}", job_id, extra2, ntime, "00000000"]})
        result = recv()
        print(f"Share result: {result.get('result')}")
